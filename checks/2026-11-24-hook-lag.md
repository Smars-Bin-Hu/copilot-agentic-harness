# 检查：公司电脑上 hook 到底慢多少

harness 里很多判断靠「两件事相差几秒」：hook 日志的一行和 Copilot 记录的一次调用算不算同一次，子 agent 的会话算不算父会话叫出来的，一个调用是不是还在跑。公司电脑上 hook 慢，这些判断就出错。改之前先量清楚。

Copilot 自己的记录（`events.jsonl`）里有每次 hook 的开始和结束（`hook.start`、`hook.end`），还有 preToolUse 是哪一次调用（调用的 id）。脚本用它们量：

- **A**：每种 hook 跑了多久（Copilot 看到的）。
- **B**：调用开始到它的 preToolUse hook 开始：同时发出的调用要排队等 hook，这里看排多久。
- **C**：调用开始到 harness 的 hook 日志写下这一行。
- **D**：Copilot 记下用户消息到 hook 日志记下这一句。负数是 hook 先记。
- **E**：父会话记下 SubagentStart 到子 agent 的第一句。harness 现在只认 10 秒内的（`CHILD_CLAIM_SECONDS`），超出的个数单独列出。

只打印时间和数量，不打印内容。

Mac 上的样子（10 个会话）：A 全在 0.6 秒以内；B 0；C 中位 0.1、最大 2.2；D 中位 0；E 最大 0.3。

## 1. 写下脚本

在 harness 文件夹里打开 PowerShell，整段粘贴：

```powershell
@'
# How late the hooks are on this computer: from Copilot's own record (hook.start / hook.end) and the harness's hook log.
# Times and counts only, no text. Run in the harness folder:  python lag_check.py [sessions]
import json
import sys
from collections import defaultdict
from pathlib import Path

root = Path.cwd()
sys.path.insert(0, str(root / ".harness" / "engine"))
from core import copilot_events, logs, registry  # noqa: E402
from modules.task import callmatch  # noqa: E402

S = copilot_events.seconds
want = int(sys.argv[1]) if len(sys.argv) > 1 else 10
spans = defaultdict(list)      # A: how long each kind of hook took, as Copilot saw it
queued = []                    # B: a tool call's start to its preToolUse hook's start (calls side by side wait in a queue)
late = defaultdict(list)       # C: the hook log line's time minus the start of the call in Copilot's record
prompt_gap = []                # D: the hook log's UserPromptSubmit minus Copilot's user.message
claim = []                     # E: a subagent's first prompt line minus the SubagentStart line of its parent


def stats(values):
    if not values:
        return "n 0"
    v = sorted(values)
    pick = lambda q: v[min(len(v) - 1, int(q * (len(v) - 1) + 0.5))]  # noqa: E731
    return "n %d | min %.1f | p50 %.1f | p90 %.1f | max %.1f" % (len(v), v[0], pick(0.5), pick(0.9), v[-1])


files = [(surface, path) for surface, path in logs.session_files(root) if surface == "vscode"]
mains, children = [], defaultdict(list)
for surface, path in files:
    rows = logs.read_lines(path)
    parent = next((r.get("parent_session_id") for r in rows[:3] if r.get("parent_session_id")), None)
    (children[parent].append(rows) if parent else mains.append((path.stem, rows)))
done = 0
for sid, rows in reversed(mains):
    ev = copilot_events.events_path(sid)
    if ev is None:
        continue
    done += 1
    if done > want:
        break
    begun, starts, messages = {}, [], []
    hook_calls = []  # (time of the preToolUse hook's start in Copilot's record, tool name)
    with open(ev, encoding="utf-8", errors="replace") as handle:
        for line in handle:
            if '"hook.' not in line and '"tool.execution_start"' not in line and '"user.message"' not in line:
                continue
            try:
                row = json.loads(line)
            except ValueError:
                continue
            kind, data, at = row.get("type"), row.get("data") or {}, S(row.get("timestamp"))
            if at is None:
                continue
            if kind == "hook.start":
                begun[data.get("hookInvocationId")] = (at, data.get("hookType"))
                if data.get("hookType") == "preToolUse":
                    for call in (data.get("input") or {}).get("toolCalls") or []:
                        hook_calls.append((at, str(call.get("name") or ""), str(call.get("id") or "")))
            elif kind == "hook.end" and data.get("hookInvocationId") in begun:
                first, hook_type = begun.pop(data.get("hookInvocationId"))
                spans[hook_type].append(at - first)
            elif kind == "tool.execution_start":
                starts.append((at, str(data.get("toolName") or ""), str(data.get("toolCallId") or "")))
            elif kind == "user.message":
                messages.append(at)
    by_id = {ident: at for at, _name, ident in starts if ident}
    queued += [hook_at - by_id[ident] for hook_at, _name, ident in hook_calls if ident in by_id]
    pre = [(S(r.get("at")), str(r.get("tool_name") or "")) for r in rows if r.get("event") == "PreToolUse" and r.get("tool_name") and not r.get("from_subagent")]
    pre = [item for item in pre if item[0] is not None]
    for index, number in callmatch.pair(pre, [(at, name) for at, name, _ident in starts]).items():
        late["all"].append(pre[index][0] - starts[number][0])
    prompts = [S(r.get("at")) for r in rows if r.get("event") == "UserPromptSubmit" and not r.get("from_subagent") and not r.get("continuation")]
    for at in prompts:
        if at is None or not messages:
            continue
        near = min(messages, key=lambda m: abs(m - at))
        if abs(near - at) <= 600:
            prompt_gap.append(at - near)
    subs = [S(r.get("at")) for r in rows if r.get("event") == "SubagentStart"]
    for child in children.get(sid, []):
        first = next((S(r.get("at")) for r in child if r.get("event") == "UserPromptSubmit"), None)
        before = [at for at in subs if at is not None and first is not None and at <= first]
        if before:
            claim.append(first - max(before))

print("V version", registry.harness_version(root), "| python", sys.version.split()[0], "| sessions", done if done <= want else want)
print("A how long a hook ran (Copilot's record), seconds")
for hook_type, values in sorted(spans.items()):
    print("   ", str(hook_type).ljust(22), stats(values))
print("B a call's start -> its preToolUse hook starts  ", stats(queued))
print("C a call's start -> the hook log line           ", stats(late["all"]))
print("D Copilot's prompt -> the hook log's prompt line", stats(prompt_gap), "(minus: the hook line came first)")
print("E SubagentStart -> the subagent's first prompt  ", stats(claim), "| over 10 s:", sum(1 for x in claim if x > 10))
'@ | Set-Content -Encoding UTF8 "$env:TEMP\lag_check.py"
```

## 怎么读（给 Claude）

- **A** 的 p90 和 max 就是 hook 的真实耗时，和 `.github/hooks/harness.json` 的超时（30 秒）比：接近超时的，Copilot 会不等 hook 就放行调用。
- **B** 大于 0 说明 hook 排队；同时发出的调用越多越大（今天 7 个编辑排了 21–28 秒）。
- **C** 的 max 决定 hook 日志和 Copilot 记录按时间配对要留多少余量。
- **D** 的范围决定一条用户消息在两份记录里配对的窗口（现在 `HOOK_EARLY_SECONDS` 30、`MATCH_SECONDS` 5）。
- **E** 有超过 10 秒的，说明子 agent 的会话可能没被认成父会话叫出来的；这会影响 gate 怎么判断子 agent 的调用。

---

## 结果（2026-11-24 发回，20 个会话）

- **A hook 本身很慢**：各种 hook 的中位数 4–8 秒，p90 10–22 秒。最长的：postToolUse 111.5 秒，preToolUse 75.9 秒，agentStop 76.9 秒。preToolUse 超过了 30 秒的超时，说明 Copilot 要么没在 30 秒时结束它，要么 hook.start 到 hook.end 里还包括了排队（见第二部分）。userPromptSubmitted 最长 30.1、userPromptTransformed 32.3，正好落在 30 秒附近，像是撞上了超时。
- **B 排队 0**：调用一开始 Copilot 就记下了 hook 开始（最多 0.4 秒）。所以慢的不是 Copilot 排队，而是我们的 Python 进程本身（启动、等锁、扫描）。
- **C hook 日志比调用开始晚**：中位 4.9 秒，p90 15.2 秒，最长 63.6 秒。现在 Agent actions 用 3 秒（`SAME_CALL_S`）判断同一批调用，必然出错。
- **D 用户消息**：hook 日志总是先记，比 Copilot 的 `user.message` 早 2.5–97.9 秒（中位 5.8 秒）。原因是 Copilot 等两个提示 hook 跑完才写这条消息。现在的窗口 `HOOK_EARLY_SECONDS` 30 秒不够。
- **E** 这 20 个会话里没有子 agent，没量到。

结论：按时间配对要改成用 Copilot 自己的时间戳（hook 收到的数据里就有），1.8.47 先做 Agent actions 这一处。

## 第二部分：超过 30 秒的 hook，Copilot 怎么处理

要知道：超时时 Copilot 有没有结束 hook、hook 算失败还是成功、调用有没有照样跑（gate 有没有被绕过）。

### 4. 写下第二个脚本

```powershell
@'
# Hooks that ran longer than the timeout (30 s): did Copilot wait for them, end them, and did the call run anyway?
# Times, kinds and the first words of an error only. Run in the harness folder:  python timeout_check.py [sessions]
import json
import sys
from collections import Counter
from pathlib import Path

root = Path.cwd()
sys.path.insert(0, str(root / ".harness" / "engine"))
from core import copilot_events, logs, redact  # noqa: E402

S = copilot_events.seconds
want = int(sys.argv[1]) if len(sys.argv) > 1 else 20
LIMIT = 30.0
sessions = [path.stem for surface, path in logs.session_files(root) if surface == "vscode"]
tally, shown, seen = Counter(), 0, 0
print("H hooks that ran over 30 s (at most 25 shown)")
for sid in reversed(sessions):
    ev = copilot_events.events_path(sid)
    if ev is None:
        continue
    seen += 1
    if seen > want:
        break
    begun, ended, slow = {}, {}, []
    with open(ev, encoding="utf-8", errors="replace") as handle:
        for line in handle:
            if '"hook.' not in line and '"tool.execution_complete"' not in line:
                continue
            try:
                row = json.loads(line)
            except ValueError:
                continue
            kind, data, at = row.get("type"), row.get("data") or {}, S(row.get("timestamp"))
            if kind == "hook.start":
                ids = [str(c.get("id") or "") for c in (data.get("input") or {}).get("toolCalls") or []]
                begun[data.get("hookInvocationId")] = (at, data.get("hookType"), ids)
            elif kind == "tool.execution_complete":
                ended[str(data.get("toolCallId") or "")] = data.get("success")
            elif kind == "hook.end" and data.get("hookInvocationId") in begun:
                first, hook_type, ids = begun.pop(data.get("hookInvocationId"))
                span = (at or 0) - (first or 0)
                output = data.get("output")
                said = (data.get("error") or {}).get("message") if isinstance(data.get("error"), dict) else data.get("error")
                denied = isinstance(output, dict) and any(str(v).startswith("Denied") for v in output.values())
                over = span > LIMIT
                tally[(hook_type, "over" if over else "under", "ok" if data.get("success") else "failed")] += 1
                if over:
                    slow.append(lambda ids=ids, sid=sid, hook_type=hook_type, span=span, data=data, denied=denied, output=output, said=said: print("   ", sid[:8], str(hook_type)[:18].ljust(18), "%.1f s" % span, "| success", data.get("success"), "| denied" if denied else "",
                          "| output keys %d" % (len(output) if isinstance(output, dict) else 0), "| tool ended", [ended.get(i) for i in ids] if ids else "-",
                          ("| error " + " ".join(redact.text(str(said)).split())[:70]) if said else ""))
    for show in slow:
        if shown < 25:
            shown += 1
            show()
print("T hooks by kind: over or under 30 s, success or not")
for key in sorted(tally):
    print("   ", " ".join(str(k) for k in key).ljust(40), tally[key])
'@ | Set-Content -Encoding UTF8 "$env:TEMP\timeout_check.py"
```

### 怎么读（给 Claude）

- **H**：每个超过 30 秒的 hook 一行：会话、种类、耗时、`success`、是否拒绝（`denied`）、`output keys`（给了几个调用的答复）、`tool ended`（那几个调用最后成功没有）、错误开头。`success False` 加上 "timed out" 一类的错误，再加上 `tool ended [True]`，就是超时后调用照样跑了：gate 被绕过。`success True` 加上有答复，说明 Copilot 等完了 hook，30 秒的超时没有生效。

## 第二部分的结果（2026-11-24 发回，20 个会话）

- **超时是真的，而且调用照样跑了。** preToolUse 有 56 次超过 30 秒：52 次 Copilot 报 `HookTimeoutError: Hook command timed out after 30 seconds`，`success False`，没有拿到答复，对应的调用全部 `tool ended True`。也就是说，这 52 次（约占 887 次的 6%）gate 没来得及判断，调用就被放行了。

- userPromptSubmitted 1 次、userPromptTransformed 3 次、agentStop 4 次超时：提示带的上下文（语言、任务）和停下时的检查，这几次都没生效。

## 第三部分：时间花在哪，被放行的调用 gate 本来怎么判

要知道：
- **M**：hook 自己的代码跑了多久（hook 日志每行的 `ms`）。
- **N**：Copilot 量到的 hook 时间减去 M，也就是启动 Python、加载代码、杀毒扫描花的时间。M 大就是我们的代码慢（等锁、读文件），N 大就是进程启动慢。
- **X**：超时被放行的那些 preToolUse，gate 后来给的答复（`allow`/`none`、`deny`、`held`）和调用跑没跑。`deny` 或 `held` 加上 `ran`，就是 gate 本来要拦、却已经跑了的调用。

### 6. 写下第三个脚本

```powershell
@'
# Part 3: where a hook's time goes, and whether a call the gate said no to ran because Copilot stopped waiting.
# Times, kinds and counts only. Run in the harness folder:  python gate_miss_check.py [sessions]
import json
import sys
from collections import Counter, defaultdict
from pathlib import Path

root = Path.cwd()
sys.path.insert(0, str(root / ".harness" / "engine"))
from core import copilot_events, logs  # noqa: E402
from modules.task import callmatch  # noqa: E402

S = copilot_events.seconds
want = int(sys.argv[1]) if len(sys.argv) > 1 else 20
inside = defaultdict(list)   # M: the time the hook spent in its own code (`ms` of the hook log), by event
outside = defaultdict(list)  # N: Copilot's span of a hook minus that: starting Python, loading, the antivirus
miss = Counter()             # X: timed-out preToolUse hooks, by what the gate answered later
names = {"preToolUse": "PreToolUse", "postToolUse": "PostToolUse", "userPromptSubmitted": "UserPromptSubmit", "agentStop": "Stop"}


def stats(values):
    if not values:
        return "n 0"
    v = sorted(values)
    pick = lambda q: v[min(len(v) - 1, int(q * (len(v) - 1) + 0.5))]  # noqa: E731
    return "n %d | min %.1f | p50 %.1f | p90 %.1f | max %.1f" % (len(v), v[0], pick(0.5), pick(0.9), v[-1])


seen = 0
for surface, path in reversed(list(logs.session_files(root))):
    if surface != "vscode":
        continue
    rows = logs.read_lines(path)
    ev = copilot_events.events_path(path.stem)
    if ev is None or any(r.get("parent_session_id") for r in rows[:3]):
        continue
    seen += 1
    if seen > want:
        break
    for r in rows:
        if isinstance(r.get("ms"), (int, float)):
            inside[str(r.get("event"))].append(r["ms"] / 1000.0)
    begun, spans = {}, []  # spans: (start, end, hook type, tool name, success, ids)
    ended = {}
    with open(ev, encoding="utf-8", errors="replace") as handle:
        for line in handle:
            if '"hook.' not in line and '"tool.execution_complete"' not in line:
                continue
            try:
                row = json.loads(line)
            except ValueError:
                continue
            kind, data, at = row.get("type"), row.get("data") or {}, S(row.get("timestamp"))
            if kind == "hook.start":
                calls = (data.get("input") or {}).get("toolCalls") or []
                begun[data.get("hookInvocationId")] = (at, data.get("hookType"), str((calls[0] if calls else {}).get("name") or ""), [str(c.get("id") or "") for c in calls])
            elif kind == "hook.end" and data.get("hookInvocationId") in begun:
                first, hook_type, name, ids = begun.pop(data.get("hookInvocationId"))
                spans.append((first, at, hook_type, name, bool(data.get("success")), ids))
            elif kind == "tool.execution_complete":
                ended[str(data.get("toolCallId") or "")] = data.get("success")
    # N: pair each Copilot span with the hook log line of the same event that ended last before Copilot's end, same tool family for a tool hook
    lines = defaultdict(list)
    for r in rows:
        if isinstance(r.get("ms"), (int, float)) and S(r.get("at")) is not None and not r.get("from_subagent"):
            lines[str(r.get("event"))].append((S(r.get("at")), str(r.get("tool_name") or ""), r))
    for first, last, hook_type, name, ok, ids in spans:
        event = names.get(hook_type)
        if event is None or first is None or last is None:
            continue
        found = [item for item in lines[event] if first - 2 <= item[0] and (not name or callmatch.family(item[1]) == callmatch.family(name))]
        if not found:
            if hook_type == "preToolUse" and not ok:
                miss["no hook log line"] += 1
            continue
        at, _tool, r = min(found, key=lambda item: item[0])
        lines[event].remove((at, _tool, r))
        if ok:
            outside[event].append(max(0.0, (last - first) - r["ms"] / 1000.0))
        elif hook_type == "preToolUse":
            answer = "held" if r.get("decision") == "deny" and isinstance(r.get("external"), dict) else str(r.get("decision") or "none")
            ran = [ended.get(i) for i in ids]
            miss[(answer, "ran" if any(x is True for x in ran) else "did not run")] += 1

print("M the hook's own code (ms of the hook log), seconds")
for event, values in sorted(inside.items()):
    print("   ", event.ljust(22), stats(values))
print("N the rest of Copilot's span (start Python, load, scan), seconds")
for event, values in sorted(outside.items()):
    print("   ", event.ljust(22), stats(values))
print("X preToolUse hooks Copilot stopped waiting for: what the gate answered later, and whether the call ran")
for key, count in sorted(miss.items(), key=lambda item: str(item[0])):
    print("   ", key, count)
'@ | Set-Content -Encoding UTF8 "$env:TEMP\gate_miss_check.py"
```

## 第三部分的结果（2026-11-24 发回，20 个会话）

- **M，hook 自己的代码**：preToolUse 中位 1.0 秒、p90 2.3 秒；postToolUse 中位 0.6 秒；最长的是 sessionStart，13.1 秒。
- **N，代码之外的时间**（起 Python、加载代码、杀毒扫描）：preToolUse 中位 5.0、p90 14.8、最长 38.8 秒；postToolUse 中位 4.4、最长 41.9 秒；最短也要 1.1–1.9 秒。**hook 的时间大约八成花在这里。**
- **X，超时被放行的 preToolUse，gate 后来给的答复**：放行（none）后调用跑了 42 次；**要人批准（held）却已经跑了 2 次**；放行但调用没跑 1 次；7 次 hook 日志里没有这一行（进程被结束前没写下）。没有一次是直接拒绝（deny）。
- 结论：提速要先治进程启动和加载代码，不是先改判断逻辑。那 2 次 held 是真正出过的事：本该等人批准的外部调用，没等批准就执行了。

## 第五部分：起进程的几秒由什么组成；那 2 次 held 是什么调用

- **S1–S3**：只起 Python；`-I -S` 起 Python（不加载 site 和环境）；起 Python 并加载 hook 的代码。各跑 8 次。
- **S4、S5**：7 个同时起（像 7 个并行编辑），加载 hook 和不加载 hook 各一次，看墙上时间。
- **F**：加载 hook 时一共加载了多少个模块，其中多少来自 `.harness/engine`，有几个已有最新的 `.pyc`（没有就每次重新编译），以及 `PYTHONDONTWRITEBYTECODE`。
- **Y**：超时那段时间里被 gate 判成 held 的外部调用：时间、工具、系统、环境、动作（读、写、跑）。只打印名字，不打印命令。

Mac 上：S1 0.02 秒，S3 0.08 秒，S4 0.12 秒。

### 8. 写下第五个脚本

### 怎么读（给 Claude）

- `pre>` 到 `pre<` 很长、而且一个接一个往后排（第一个 5 秒、第二个 10 秒…）：hook 在排队，一个等一个。我们的 hook 每次都要拿同一个会话的锁，同时来的 hook 就排成一列。
- `pre<` 到 `post>` 很长：工具自己慢（少见）。
- `post>` 到 `post<` 很长：postToolUse hook 也在排队，调用要等它结束才算结束。
- 一批的总时间约等于「hook 个数 × 每个 hook 的时间」：就是排队，提速要先解决锁。

## 第九部分的结果（2026-11-24 发回）

## 第十三部分：1.8.58 之后，hook 快了多少

1.8.58 一次做了三件事（试验见第十、十一部分）：

- **不经 PowerShell**：每个 hook 用 `exec: "python"`、`args: ["-S", ".harness/engine/launch.py"]`、`cwd: "."` 直接起 Python，没有 shell。`python` 要是 PATH 上真的 `python.exe`（`exec` 只认 .exe 和 .com）。
- **判断放进看板**：看板开着时，看板进程里有一个判断服务（只听 127.0.0.1）。hook 进程只是个小客户端：读 `.harness/runtime/hookserve.json`，和服务握手（服务要先证明有口令），把 Copilot 给的数据发过去，等「开始」那一行（`wait` 秒，约 3 秒乘看板量到的节奏系数），再等答复，最多 20 秒。任何一点不对（看板没开、服务答得慢、看板里的代码和磁盘上的不是同一份、TEMP 不同、Python 版本不同……），客户端就在自己进程里跑完整的判断，和 1.8.56 一样。hook 日志每行记下 `via`（`service` 或 `local`）、`why`（为什么是 local）、`ms`（判断本身花的毫秒）、`nonce`；客户端另在 `.harness/runtime/logs/hookclient.jsonl` 留一行痕迹。

- `wait`（脚本的 `H` 行，也在 `hookserve.json` 里）：约 3 秒乘看板量到的节奏系数。如果多数 `local` 是 `wait`，而 `wait` 的值不大（3–4 秒），说明这台电脑上服务回「开始」要更久（看板忙、同一时间多个 hook，服务一次只判断一个）；要把 `wait` 调大，或者看看板进程为什么忙。若 `wait` 已经很大（15 秒以上）还是 `wait`：服务真的卡住了。
- 最后，`H` 行写着 `hookserve.json is not there`：服务没起（设置里关了 `hook_service`，或启动失败）。

## 第十四部分：正在跑的任务，实时看（1.8.58）

起因：真任务里一批二十多个编辑，Live 上很多行绿点、时间一样（例如都是 1 min 14 s），没有 +/- 数。看起来像卡死。

我的判断（Mac 上 Copilot 的记录确认过顺序）：一批调用里，Copilot **一个一个**跑 post hook，每个调用要等自己的 post hook 结束才算完成（`tool.execution_complete` 在 post hook 之后）。所以一批 20 个编辑：preToolUse 一次管整批，之后 post hook 排队；第 N 个编辑要等前面 N-1 个 post hook。绿点、同一个时间 = 同一批一起开始、还在排队，不是卡死。如果每个 post hook 约 3.5 秒，20 个就是约 70 秒，正好是 1 min 14 s 这个量级。

这个脚本在任务跑的时候看：哪些调用没完成、是在跑 post hook 还是在等；每种 hook 现在多少秒；最近几批从开始到结束多久；hook 日志里谁判断的。约 3 分钟，拍 1–2 张照。

### 40. 写下脚本

在 harness 文件夹的 PowerShell 里：

```powershell
@'
# Part 14: a running task, live. Is a call stuck, or waiting in the queue of post hooks? How long does each hook take now (1.8.58)?
# Copilot's own record (events.jsonl) and the harness's hook log. Times, counts and tool names only: no text, no paths.
# Run in the harness folder while the task runs:  python live_check.py [session id start]   (default: the chat whose record changed last)
import json
import sys
import time
from collections import Counter, defaultdict
from pathlib import Path

root = Path.cwd()
sys.path.insert(0, str(root / ".harness" / "engine"))
from core import copilot_events, logs, registry  # noqa: E402

S = copilot_events.seconds
now = time.time()
arg = sys.argv[1] if len(sys.argv) > 1 else ""


def stats(values):
    if not values:
        return "n   0"
    v = sorted(values)
    return "n %3d | p50 %6.2f s | max %6.2f s" % (len(v), v[len(v) // 2], v[-1])


states = copilot_events.home() / "session-state"
records = []
for folder in states.iterdir() if states.is_dir() else []:
    path = folder / "events.jsonl"
    if path.is_file() and (not arg or folder.name.startswith(arg)):
        records.append((path.stat().st_mtime, folder.name, path))
if not records:
    sys.exit("no Copilot record found" + (" for " + arg if arg else ""))
changed, sid, path = max(records)
print("V version", registry.harness_version(root), "| session", sid[:8], "| record last changed %.0f s ago" % (now - changed))

calls = {}                 # toolCallId -> dict(name, start, done, post_start, post_end)
hooks = {}                 # hookInvocationId -> (type, start, call id)
spans = defaultdict(list)  # hook type -> seconds, finished hooks
open_hooks = []
with open(path, encoding="utf-8", errors="replace") as handle:
    for line in handle:
        if '"hook.' not in line and '"tool.execution_' not in line:
            continue
        try:
            row = json.loads(line)
        except ValueError:
            continue
        kind, data, at = row.get("type"), row.get("data") or {}, S(row.get("timestamp"))
        if at is None or row.get("agentId"):
            continue
        if kind == "tool.execution_start":
            calls[str(data.get("toolCallId"))] = {"name": str(data.get("toolName") or ""), "start": at}
        elif kind == "tool.execution_complete":
            call = calls.get(str(data.get("toolCallId")))
            if call is not None:
                call["done"] = at
        elif kind == "hook.start":
            given = data.get("input") if isinstance(data.get("input"), dict) else {}
            ident = str(given.get("toolCallId") or "")
            hooks[data.get("hookInvocationId")] = (str(data.get("hookType")), at, ident)
            if data.get("hookType") == "postToolUse" and ident in calls:
                calls[ident]["post_start"] = at
        elif kind == "hook.end" and data.get("hookInvocationId") in hooks:
            hook_type, first, ident = hooks.pop(data.get("hookInvocationId"))
            spans[hook_type].append(at - first)
            if hook_type == "postToolUse" and ident in calls:
                calls[ident]["post_end"] = at

print("O calls not finished yet (no tool.execution_complete):")
waiting = [c for c in calls.values() if "done" not in c]
if not waiting:
    print("    none: nothing is running or stuck")
for state, test in (("in its post hook now", lambda c: "post_start" in c and "post_end" not in c),
                    ("waiting for its post hook to start", lambda c: "post_start" not in c),
                    ("post hook done, tool not finished", lambda c: "post_end" in c)):
    group = [c for c in waiting if test(c)]
    if group:
        ages = [now - c["start"] for c in group]
        print("   ", state.ljust(36), "%3d calls | oldest %6.0f s | tools %s" % (len(group), max(ages), dict(Counter(c["name"] for c in group))))
for hook_type, first, _ident in hooks.values():
    print("    open hook:", hook_type.ljust(20), "started %6.0f s ago" % (now - first))

print("H every finished hook of this session, Copilot's seconds from hook.start to hook.end (before 1.8.58: preToolUse p50 4-8 s, max 75.9 s)")
for hook_type, values in sorted(spans.items()):
    print("   ", hook_type.ljust(22), stats(values), "| over 120 s: %d" % sum(1 for v in values if v > 120))

# batches: 3+ calls Copilot started within 2 s of each other
ordered = sorted(calls.values(), key=lambda c: c["start"])
batches, current = [], []
for call in ordered:
    if current and call["start"] - current[-1]["start"] > 2:
        if len(current) >= 3:
            batches.append(current)
        current = []
    current.append(call)
if len(current) >= 3:
    batches.append(current)
print("B the newest 5 batches (3+ calls started together): calls | tools | first start to last finish | before the first post hook | post hooks each p50/max | still open")
for batch in batches[-5:]:
    first = batch[0]["start"]
    ends = [c["done"] for c in batch if "done" in c]
    posts = [c["post_end"] - c["post_start"] for c in batch if "post_end" in c and "post_start" in c]
    post_starts = [c["post_start"] for c in batch if "post_start" in c]
    took = ("%6.1f s" % (max(ends) - first)) if len(ends) == len(batch) else ("running %5.0f s" % (now - first))
    before = ("%5.1f s" % (min(post_starts) - first)) if post_starts else "  -  "
    each = ("%.2f/%.2f s" % (sorted(posts)[len(posts) // 2], max(posts))) if posts else "-"
    print("   ", time.strftime("%H:%M:%S", time.localtime(first)), "| %2d | %s | %s | %s | %s | %d" % (
        len(batch), ",".join("%s x%d" % kv for kv in Counter(c["name"][:22] for c in batch).items()), took, before, each, len(batch) - len(ends)))
```
